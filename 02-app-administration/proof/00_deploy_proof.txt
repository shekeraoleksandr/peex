===== Customized configuration =====
$ cat /etc/peex-web/nginx-peex.conf
# Customized standalone nginx configuration for the PeEx demo service.
# Runs entirely under the unprivileged service user (systemd sets User=peexweb).
# Binds to a high port so no root is needed.

worker_processes auto;
pid /run/peex-web/nginx.pid;
error_log /var/log/peex-web/error.log warn;

events {
    worker_connections 1024;
}

http {
    include      /etc/nginx/mime.types;
    default_type application/octet-stream;

    access_log /var/log/peex-web/access.log;

    client_body_temp_path /var/lib/peex-web/tmp/client;
    proxy_temp_path       /var/lib/peex-web/tmp/proxy;
    fastcgi_temp_path     /var/lib/peex-web/tmp/fastcgi;
    uwsgi_temp_path       /var/lib/peex-web/tmp/uwsgi;
    scgi_temp_path        /var/lib/peex-web/tmp/scgi;

    sendfile      on;
    keepalive_timeout 65;
    server_tokens off;

    server {
        listen 8080;
        server_name _;
        root  /var/www/peex-web;
        index index.html;

        location = /healthz {
            default_type text/plain;
            return 200 "ok\n";
        }

        location / {
            try_files $uri $uri/ =404;
        }
    }
}

===== systemd unit =====
$ systemctl cat peex-web
# /etc/systemd/system/peex-web.service
[Unit]
Description=PeEx demo web service (nginx, least-privilege)
Documentation=file:/opt/peex-web/README.md
After=network-online.target
Wants=network-online.target

[Service]
Type=forking
User=peexweb
Group=peexweb
PIDFile=/run/peex-web/nginx.pid

# systemd creates and owns these dirs for the service user on every start
RuntimeDirectory=peex-web
LogsDirectory=peex-web
StateDirectory=peex-web

ExecStartPre=/usr/sbin/nginx -t -c /etc/peex-web/nginx-peex.conf
ExecStart=/usr/sbin/nginx -c /etc/peex-web/nginx-peex.conf
ExecReload=/usr/sbin/nginx -s reload -c /etc/peex-web/nginx-peex.conf
ExecStop=/usr/sbin/nginx -s quit -c /etc/peex-web/nginx-peex.conf
Restart=on-failure
RestartSec=3

# --- Hardening / principle of least privilege ---
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=true
ProtectKernelTunables=true
ProtectControlGroups=true
ReadWritePaths=/var/log/peex-web /var/lib/peex-web /run/peex-web
CapabilityBoundingSet=
AmbientCapabilities=

[Install]
WantedBy=multi-user.target
